function stableFingerprintValue(value) {
  return JSON.stringify(value, (_key, inner) => {
    if (inner && typeof inner === "object" && !Array.isArray(inner)) {
      const sorted = {};
      for (const key4 of Object.keys(inner).sort()) sorted[key4] = inner[key4];
      return sorted;
    }
    return inner;
  }) ?? "undefined";
}
function defineMcpServerSpec(build, surface) {
  return { fingerprint: stableFingerprintValue(surface), build };
}
function mcpServerSpecsAreCompatible(previous, next) {
  const previousNames = previous ? Object.keys(previous) : [];
  const nextNames = next ? Object.keys(next) : [];
  if (previousNames.length !== nextNames.length) return false;
  for (const name of nextNames) {
    const before = previous?.[name];
    const after = next?.[name];
    if (!before || !after) return false;
    if (before.fingerprint !== after.fingerprint) return false;
  }
  return true;
}
function buildMcpServerConfigMap(specs) {
  if (!specs) return;
  const entries = Object.entries(specs);
  if (entries.length === 0) return;
  const result = {};
  for (const [name, spec] of entries) result[name] = spec.build();
  return result;
}
